e.postData.contents : '{}'; const body = JSON.parse(rawBody); const name = String(body.name || '').trim().slice(0, 80); const wish = String(body.wish || '').trim().slice(0, 500); const allowedStatus = ['Hadir', 'Ragu-ragu', 'Tidak Hadir']; const attendance = allowedStatus.indexOf(body.attendance) !== -1 ? body.attendance : 'Hadir'; const guests = attendance === 'Tidak Hadir' ? 0 : Math.max(1, Math.min(20, Number(body.guests) || 1)); if (!name || !wish) { return createJsonResponse_({ status: 'error', message: 'Nama tamu dan ucapan tidak boleh kosong.' }); } const id = 'rsvp_' + Date.now(); const timestampWIB = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'dd MMM yyyy, HH:mm') + ' WIB'; // Simpan baris baru ke tab RSVP_Ucapan rsvpSheet.appendRow([id, timestampWIB, name, attendance, guests, wish]); return createJsonResponse_({ status: 'success', data: { id: id, createdAt: timestampWIB, name: name, attendance: attendance, guests: guests, wish: wish } }); } catch (err) { return createJsonResponse_({ status: 'error', message: err.message || 'Gagal menyimpan data ke Google Sheets.' }); } finally { lock.releaseLock(); } }
