/** * ============================================================================ * GOOGLE APPS SCRIPT - DATABASE UNDANGAN PERNIKAHAN DIGITAL (OPSI A) * ============================================================================ * Cara Pemasangan: * 1. Buka Google Sheets baru di https://sheets.new * 2. Klik menu: Ekstensi (Extensions) -> Apps Script * 3. Hapus seluruh kode bawaan, lalu tempel (paste) seluruh kode ini. * 4. Klik ikon Simpan (Ctrl+S), lalu jalankan fungsi "setupDatabase" sekali * dan berikan izin akses (Review Permissions -> Allow). * 5. Klik tombol "Terapkan" (Deploy) -> "Deployment baru" (New deployment). * 6. Pilih jenis: "Aplikasi web" (Web app): * - Jalankan sebagai (Execute as): Saya (Me) * - Yang memiliki akses (Who has access): Siapa saja (Anyone) * 7. Klik "Terapkan", lalu salin "URL Aplikasi Web" yang muncul. * 8. Tempel URL tersebut ke variabel `googleScriptUrl` di `undangan_pernikahan.html`. * ============================================================================ */ const SHEET_NAMES = { WEDDING: 'Data_Pengantin', GUESTS: 'Daftar_Tamu', RSVP: 'RSVP_Ucapan' }; /** * Inisialisasi otomatis 3 tab database di Google Sheets jika belum ada. * Anda bisa menjalankan fungsi ini secara manual dari editor Apps Script * agar struktur tabel langsung terbentuk dengan rapi. */ function setupDatabase() { const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Tab Data_Pengantin (Database Profil & Pengaturan Acara) let weddingSheet = ss.getSheetByName(SHEET_NAMES.WEDDING); if (!weddingSheet) { weddingSheet = ss.insertSheet(SHEET_NAMES.WEDDING); const weddingHeaders = [['Kunci', 'Nilai', 'Keterangan']]; const weddingDefaults = [ ['groom_name', 'Ardiansyah Pratama', 'Nama mempelai pria'], ['bride_name', 'Nadia Larasati', 'Nama mempelai wanita'], ['event_date_iso', '2027-02-20T08:00:00+07:00', 'Tanggal & jam mulai acara (ISO)'], ['venue_name', 'The Dharmawangsa Jakarta', 'Nama gedung / lokasi'], ['venue_address', 'Jl. Brawijaya Raya No. 26, Kebayoran Baru, Jakarta Selatan', 'Alamat lengkap lokasi'] ]; weddingSheet.getRange(1, 1, 1, 3).setValues(weddingHeaders); weddingSheet.getRange(2, 1, weddingDefaults.length, 3).setValues(weddingDefaults); styleHeaderRow_(weddingSheet, 3); } // 2. Tab Daftar_Tamu (Database Daftar Tamu Undangan untuk Kirim Link) let guestSheet = ss.getSheetByName(SHEET_NAMES.GUESTS); if (!guestSheet) { guestSheet = ss.insertSheet(SHEET_NAMES.GUESTS); const guestHeaders = [['No', 'Nama Tamu', 'Kategori', 'No WhatsApp', 'Kuota', 'Status Kirim']]; const guestExamples = [ [1, 'Budi Santoso & Pasangan', 'VIP / Sahabat', '6281234567890', 2, 'Belum Dikirim'], [2, 'Keluarga Besar Bapak Arifin', 'Keluarga', '6281298765432', 4, 'Sudah Dikirim'] ]; guestSheet.getRange(1, 1, 1, 6).setValues(guestHeaders); guestSheet.getRange(2, 1, guestExamples.length, 6).setValues(guestExamples); styleHeaderRow_(guestSheet, 6); } // 3. Tab RSVP_Ucapan (Database Otomatis Konfirmasi Kehadiran & Buku Tamu) let rsvpSheet = ss.getSheetByName(SHEET_NAMES.RSVP); if (!rsvpSheet) { rsvpSheet = ss.insertSheet(SHEET_NAMES.RSVP); const rsvpHeaders = [['ID', 'Waktu', 'Nama Tamu', 'Kehadiran', 'Jumlah Tamu', 'Ucapan & Doa']]; rsvpSheet.getRange(1, 1, 1, 6).setValues(rsvpHeaders); styleHeaderRow_(rsvpSheet, 6); } return { ss, weddingSheet, guestSheet, rsvpSheet }; } /** * Helper untuk merapikan tampilan baris header pada Google Sheets. */ function styleHeaderRow_(sheet, colCount) { const headerRange = sheet.getRange(1, 1, 1, colCount); headerRange .setFontWeight('bold') .setBackground('#F3EEE6') .setFontColor('#262320'); sheet.setFrozenRows(1); } /** * Helper untuk mengembalikan respons JSON standar yang aman dibaca browser. */ function createJsonResponse_(payload) { return ContentService .createTextOutput(JSON.stringify(payload)) .setMimeType(ContentService.MimeType.JSON); } /** * ============================================================================ * ENDPOINT GET: Mengambil Data Pengantin & Daftar RSVP/Ucapan dari Sheets * ============================================================================ */ function doGet(e) { try { const { weddingSheet, rsvpSheet } = setupDatabase(); // Ambil data profil pengantin (key-value) const weddingRows = weddingSheet.getDataRange().getValues(); const wedding = {}; for (let i = 1; i < weddingRows.length; i++) { const key = String(weddingRows[i][0] || '').trim(); const val = String(weddingRows[i][1] || '').trim(); if (key) wedding[key] = val; } // Ambil seluruh data RSVP & Ucapan const rsvpRows = rsvpSheet.getDataRange().getValues(); const wishes = []; // Mulai dari indeks 1 (lewati baris header), urutkan terbaru di atas for (let i = rsvpRows.length - 1; i >= 1; i--) { const row = rsvpRows[i]; const name = String(row[2] || '').trim(); const wish = String(row[5] || '').trim(); if (!name && !wish) continue; wishes.push({ id: String(row[0] || `row_${i}`), createdAt: String(row[1] || ''), name: name, attendance: String(row[3] || 'Hadir'), guests: Number(row[4]) || 0, wish: wish }); } return createJsonResponse_({ status: 'success', wedding: wedding, wishes: wishes }); } catch (err) { return createJsonResponse_({ status: 'error', message: err.message || 'Gagal membaca data dari Google Sheets.' }); } } /** * ============================================================================ * ENDPOINT POST: Menyimpan Konfirmasi RSVP & Ucapan Baru ke Google Sheets * ============================================================================ */ function doPost(e) { const lock = LockService.getScriptLock(); try { // Kunci antrean penulisan hingga 10 detik agar data tidak saling menimpa lock.waitLock(10000); const { rsvpSheet } = setupDatabase(); const rawBody = e && e.postData && e.postData.contents ? e.postData.contents : '{}'; const body = JSON.parse(rawBody); const name = String(body.name || '').trim().slice(0, 80); const wish = String(body.wish || '').trim().slice(0, 500); const allowedStatus = ['Hadir', 'Ragu-ragu', 'Tidak Hadir']; const attendance = allowedStatus.includes(body.attendance) ? body.attendance : 'Hadir'; const guests = attendance === 'Tidak Hadir' ? 0 : Math.max(1, Math.min(20, Number(body.guests) || 1)); if (!name || !wish) { return createJsonResponse_({ status: 'error', message: 'Nama tamu dan ucapan tidak boleh kosong.' }); } const id = 'rsvp_' + Date.now(); const timestampWIB = Utilities.formatDate(new Date(), 'Asia/Jakarta', 'dd MMM yyyy, HH:mm') + ' WIB'; // Simpan baris baru ke tab RSVP_Ucapan rsvpSheet.appendRow([id, timestampWIB, name, attendance, guests, wish]); return createJsonResponse_({ status: 'success', data: { id: id, createdAt: timestampWIB, name: name, attendance: attendance, guests: guests, wish: wish } }); } catch (err) { return createJsonResponse_({ status: 'error', message: err.message || 'Gagal menyimpan data ke Google Sheets.' }); } finally { lock.releaseLock(); } }
